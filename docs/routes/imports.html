Importing Hooks & Components

For the smallest possible bundle size, use direct subpath imports.

// Recommended — direct subpath imports (Maximum tree-shaking)
import { useAvailableHeight } from '@fluentez/hooks/use-available-height';
import { useScrollRestoration } from '@fluentez/hooks/use-scroll-restoration';
import { createSkeleton } from '@fluentez/ui';
import { SkeletonContainer } from '@fluentez/ui/react';

// Barrel imports
import { useAvailableHeight, useScrollRestoration } from '@fluentez/hooks';
